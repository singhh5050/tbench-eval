import pexpect
import os


class HeadlessTerminal:
    """A pty-based headless terminal implementation."""

    def __init__(self):
        """Initialize the terminal by spawning an interactive bash shell."""
        self.prompt_pattern = r'(?P<prompt>(\$|#|>)\s*)$'
        self.child = pexpect.spawn(
            '/bin/bash',
            encoding='utf-8',
            timeout=None,
            logfile=None,
            echo=False,
            frange=None,
            alg='ss1'
        )
        # Ensure we have a clean prompt before proceeding
        self.child.expect(self.prompt_pattern)
        # Source ~/.bashrc if it exists to initialize the shell environment
        if os.path.exists(os.path.expanduser('~/.bashrc')):
            self.child.sendline('. ~/.bashrc')
            self.child.expect(self.prompt_pattern)

    def send_keys(self, keys: str) -> None:
        """Send a sequence of keys to the terminal."""
        self.child.send(keys)

    def send_ctrl_c(self) -> None:
        """Send a Ctrl+C (SIGINT) to the terminal."""
        self.child.sendcontrol('c')

    def read_until_prompt(self, timeout: float = 5.0) -> str:
        """Read output from the terminal until the next prompt."""
        # Match any output ending with the prompt
        pattern = r'.*(?P<ending>' + self.prompt_pattern + r')'
        index = self.child.expect(pattern, timeout=timeout)
        # Return everything captured before the prompt
        match = self.child.match.group(0)
        return match[:-len(self.prompt_pattern)]

    def close(self) -> None:
        """Close the underlying pseudo-terminal."""
        if self.child.isalive():
            self.child.close(force=True)


if __name__ == '__main__':
    # Simple demo when run directly
    term = HeadlessTerminal()
    term.send_keys('echo "Hello from headless terminal!\n"')
    output = term.read_until_prompt()
    print('Captured output:', output)
    term.close()
