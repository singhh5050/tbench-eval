def write_accounts(accounts):
    records = []
    for acc_id, data in accounts.items():
        name = data['name'].ljust(20)[:20]
        balance = f"{data['balance']:010}"
        records.append(acc_id.encode('ascii') + name.encode('ascii') + balance.encode('ascii'))

    with open('data/ACCOUNTS.DAT', 'wb') as f:
