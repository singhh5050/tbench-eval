" Recursive function to emit split commands based on winlayout()
function! s:EmitLayout(layout, out) abort
  if type(a:layout) == type([]) && len(a:layout) == 2 && a:layout[0] ==# 'leaf'
    " leaf node, nothing to do (window already exists)
    return
  endif
  if type(a:layout) == type([]) && len(a:layout) == 2 && a:layout[0] ==# 'col'
    " column split: first window is left, then split vertically for each subsequent
    for l:i in range(1, len(a:layout[1]))
      if l:i > 1
        call writefile(['vertical split'], a:out, 'a')
      endif
      call s:EmitLayout(a:layout[1][l:i-1], a:out)
    endfor
    return
  endif
  if type(a:layout) == type([]) && len(a:layout) == 2 && a:layout[0] ==# 'row'
    " row split: first window is top, then split horizontally for each subsequent
    for l:i in range(1, len(a:layout[1]))
      if l:i > 1
        call writefile(['split'], a:out, 'a')
      endif
      call s:EmitLayout(a:layout[1][l:i-1], a:out)
    endfor
    return
  endif
endfunction
