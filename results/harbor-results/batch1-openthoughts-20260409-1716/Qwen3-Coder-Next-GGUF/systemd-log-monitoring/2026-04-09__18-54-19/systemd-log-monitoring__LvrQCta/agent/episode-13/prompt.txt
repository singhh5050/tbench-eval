=== Monitoring Script ===
#!/bin/bash

SERVICE="myapp"
TIME_WINDOW="5 minutes"

# Get logs from last 5 minutes in JSON format
LOGS=$(journalctl -u $SERVICE --since "5 minutes ago" --json 2>/dev/null)

# Count restart events (Started messages) - ensure we get a clean number
RESTARTS=$(echo "$LOGS" | grep '"MESSAGE":".*Started"' 2>/dev/null | wc -l | tr -d ' ')

# Count error events
ERRORS=$(echo "$LOGS" | grep '"level":"ERROR"' 2>/dev/null | wc -l | tr -d ' ')

# Default to 0 if empty
RESTARTS=${RESTARTS:-0}
ERRORS=${ERRORS:-0}

# Detect restart loop: more than 3 restarts in 5 minutes
if [ "$RESTARTS" -gt 3 ]; then
    RESTART_LOOP="true"
else
    RESTART_LOOP="false"
fi

# Generate timestamp
TIMESTAMP=$(date -u +"%Y-%m-%dT%H:%M:%SZ")

# Output JSON report
cat << EOF
{
  "service": "$SERVICE",
  "time_window": "$TIME_WINDOW",
  "restarts": $RESTARTS,
  "errors": $ERRORS,
  "restart_loop_detected": $RESTART_LOOP,
  "timestamp": "$TIMESTAMP"
}
EOF
