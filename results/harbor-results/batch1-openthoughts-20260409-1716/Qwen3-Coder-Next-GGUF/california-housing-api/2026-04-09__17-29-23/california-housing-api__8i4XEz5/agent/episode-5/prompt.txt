@app.post("/predict")
async def predict_single(features: list):
    if best_model is None:
        return JSONResponse(content={"error": "Model not loaded"}, status_code=503)

    features_array = np.array(features).reshape(1, -1)
    prediction = best_model.predict(features_array)[0]

    return {
        "prediction": float(prediction),
        "model_name": model_name
    }

@app.post("/predict-batch")
async def predict_batch(features_list: list):
    if best_model is None:
        return JSONResponse(content={"error": "Model not loaded"}, status_code=503)

    features_array = np.array(features_list)
    predictions = best_model.predict(features_array)

    return {
        "predictions": [float(p) for p in predictions],
        "model_name": model_name,
        "count": len(predictions)
    }
