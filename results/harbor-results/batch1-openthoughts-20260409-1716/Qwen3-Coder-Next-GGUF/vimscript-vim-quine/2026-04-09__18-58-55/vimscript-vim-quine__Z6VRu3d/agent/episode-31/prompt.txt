Current Terminal Screen:
                    let script = script . ind . 'edit ' . fnameescape(win.bufname) . '\n'
                endif
            else
                let script = script . ind . '" Window ' . (win_idx + 1) . ' - buffer: ' . win.bufname . '\n'

                if win.split_type == 'vsplit'
                    let script = script . ind . 'vsplit\n'
                elseif win.split_type == 'split'
                    let script = script . ind . 'split\n'
                endif

                if win.bufname != ''
                    let script = script . ind . 'edit ' . fnameescape(win.bufname) . '\n'
                endif
            endif

            let script = script . ind . '\n'
        endfor
    endfor

    let script = script . ind . '" Restore active tab\n'
    let script = script . ind . 'if ' . tab_count . ' > 1\n'
    let script = script . ind1 . 'tabnext ' . a:layout.active_tab . '\n'
    let script = script . ind . 'endif\n'

    return script
endfunction

    let script = s:GenerateLayoutScript(layout, 0)

    call writefile(split(script, '\n'), a:path)

    echo 'Layout saved to: ' . a:path
endfunction
