class Sigmoid(BaseLayer):
    """Sigmoid activation function."""

    def __init__(self):
        super().__init__()

    def forward(self, x: np.ndarray) -> np.ndarray:
        """
        Args:
            x: input array of any shape, dtype float64

        Returns:
            output of same shape, dtype float64
            Formula: 1 / (1 + exp(-x))
            Numerically stable for large positive/negative values
        """
        # Numerically stable sigmoid
        # For positive x: 1 / (1 + exp(-x))
        # For negative x: exp(x) / (1 + exp(x))
        positive_mask = x >= 0
        negative_mask = ~positive_mask

        result = np.zeros_like(x, dtype=np.float64)

        # For positive values
        pos_x = x[positive_mask]
        result[positive_mask] = 1.0 / (1.0 + np.exp(-pos_x))

        # For negative values
        neg_x = x[negative_mask]
        exp_x = np.exp(neg_x)
        result[negative_mask] = exp_x / (1.0 + exp_x)

        return result
